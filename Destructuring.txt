object dan array destructuring:
1. array
const name = ["farid", "nubaili", "alkatiri"]
const [firstName, lastName, ...others] = name

others = akan menjadi ["alkatiri"]

2. objek, (nama key harus sesuai)
const person = {
	firstName: "farid", 
	lastName: "nubaili", 
	age: 23
}
const [firstName, lastName, ...others] = person

others = akan menjadi {
	age: 23
}


untuk nested:
{name, address:{name, country}, contact}, mengambil field name dan country dari address
